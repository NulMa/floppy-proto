# 감각 수치 — 게임 안에서 F5 를 누르면 재빌드 없이 다시 읽는다.
# "이름 값" 한 줄에 하나. # 은 주석.
#
# 여기 있는 숫자는 전부 기획/디렉팅이 정할 것들이다.
# 코드를 안 건드리고 이 파일만 고쳐서 감각을 맞춰라.

# --- 기본 이동 ---
gravity          900
jumpVel          -262      # 음수. 현재 정점 38px = 2.4타일
jumpCut          0.45      # 점프키를 일찍 떼면 상승속도를 이만큼 깎는다
maxFall          420
accelGround      1500
accelAir         900
maxSpeed         128       # px/s. 16px 타일 기준 초당 8타일
frictionGround   1700
frictionAir      320
coyoteTime       0.09      # 발판에서 떨어진 뒤에도 점프가 먹히는 시간
jumpBuffer       0.10      # 착지 전에 미리 누른 점프를 기억하는 시간

# --- jump2.exe (5칸) 2단 점프 ---
jump2Mult        0.92      # 2단 점프 세기 (1단 대비). 1.0 이면 똑같은 높이

# --- dash.exe (4칸) 공중 대시 ---
dashSpeed        340
dashTime         0.18
dashCooldown     0.35

# --- glide.exe (4칸) 활공 ---
glideFall        70        # 활공 중 최대 낙하속도. 낮을수록 오래 뜬다

# --- grip.exe (4칸) 벽 타기 ---
gripSlide        60        # 벽에 붙어 미끄러지는 속도
gripJumpX        190       # 벽 점프 수평 세기
gripJumpY        -250      # 벽 점프 수직 세기
gripStick        0.12      # 벽 점프 직후 방향키가 안 먹는 시간 (벽에 도로 붙는 것 방지)

# --- boot.exe (3칸) 점프력 증가 ---
bootMult         1.20      # 55px = 3.4타일

# --- 점프패드 (능력 아님, 지형 기믹) ---
# 정해진 속도로 그냥 쏘면 조작이 아니라 사고처럼 느껴진다.
# 점프키를 누른 채 밟으면 더 높이 튄다 — 개입할 여지가 있어야 조작이 된다.
padVel           -390      # 기본. 정점 약 85px = 5.3타일
padVelHold       -520      # 점프키를 누른 채. 정점 약 150px = 9.4타일

# --- 사망 연출 ---
deathFreeze      0.45      # 죽고 나서 멈춰 있는 시간.
                           # 곧바로 되살리면 가시였는지 파동이었는지도 못 본다

# --- 포맷 파동 ---
# 속도는 스테이지마다 levels.c 에 있다 (22 ~ 52px/s).
# 여기 둘은 전역이다.
formatLead       280       # 터미널을 떠날 때 파동이 뒤에 잡히는 거리
                           # 세로 구간에서는 가로 진행이 0인데 파동은 계속 온다.
                           # 200 이면 12칸 굴뚝(약 3.3초)에서 여유가 1.3배뿐이었다
formatScale      1.0       # 전 스테이지 파동 속도에 곱하는 배수. 난이도 일괄 조정용
